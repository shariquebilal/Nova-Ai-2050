export default async function handler(req, res) { const videoUrl = req.query.url; if (!videoUrl) { return res.status(400).json({ success: false, message: "Bhai, link toh daalo!" }); } try { // VERCEL SERVER direct API se baat kar raha hai (No Browser Block) const response = await fetch("https://api.cobalt.tools/api/json", { method: "POST", headers: { "Accept": "application/json", "Content-Type": "application/json", "User-Agent": "NOVA-AI-Server/1.0" }, body: JSON.stringify({ url: videoUrl }) }); const data = await response.json(); if (data.url) { // API ne raw mp4 link de di res.status(200).json({ success: true, downloadUrl: data.url }); } else if (data.text) { // API ka koi error message res.status(400).json({ success: false, message: data.text }); } else { res.status(400).json({ success: false, message: "API ne reject kar diya." }); } } catch (error) { res.status(500).json({ success: false, message: "Backend Server Crash ho gaya!" }); } }
